/**
 * Cedar Creek Care Home (Demo #2) — Interactive Logic
 * 1. Quick-Pitch URL Personalizer (?name=...&city=...&phone=...&license=...)
 * 2. Scroll-aware Transparent Navbar
 * 3. Accessible Mobile Menu Drawer
 * 4. Hero Glass Tour Form Handler
 * 5. FAQ Accordion
 */

document.addEventListener('DOMContentLoaded', () => {
  initQuickPitchPersonalizer();
  initScrollNavbar();
  initMobileNavigation();
  initHeroTourForm();
  initFaqAccordion();
});

/* 1. Quick-Pitch Personalizer */
function initQuickPitchPersonalizer() {
  const params = new URLSearchParams(window.location.search);
  const name = params.get('name');
  const city = params.get('city');
  const phone = params.get('phone');
  const license = params.get('license');
  const hideBar = params.get('hidebar') === '1';

  if (!name && !city && !phone && !license) return;

  if (name) {
    document.querySelectorAll('[data-brand-name]').forEach((el) => {
      el.textContent = name;
    });
    document.title = document.title.replace('Cedar Creek Care Home', name);
  }

  if (city) {
    document.querySelectorAll('[data-brand-city]').forEach((el) => {
      el.textContent = city;
    });
  }

  if (phone) {
    const digitsOnly = phone.replace(/\D/g, '');
    document.querySelectorAll('[data-brand-phone]').forEach((el) => {
      el.textContent = phone;
    });
    document.querySelectorAll('[data-brand-phone-link]').forEach((el) => {
      el.setAttribute('href', `tel:${digitsOnly}`);
    });
  }

  if (license) {
    document.querySelectorAll('[data-brand-license]').forEach((el) => {
      el.textContent = license;
    });
  }

  if (!hideBar) {
    const banner = document.createElement('div');
    banner.className =
      'fixed bottom-4 right-4 z-50 flex items-center gap-3 rounded-full bg-[#1E1A17] px-4 py-2 text-xs text-white shadow-xl border border-white/15';
    banner.innerHTML = `
      <span class="inline-block h-2 w-2 rounded-full bg-emerald-400"></span>
      <span>Pitch Preview: <strong>${escapeHtml(name || 'Custom Home')}</strong></span>
      <a href="${window.location.pathname}" class="underline text-[#C8A078] ml-1">Reset</a>
    `;
    document.body.appendChild(banner);
  }
}

function escapeHtml(str) {
  const div = document.createElement('div');
  div.textContent = str;
  return div.innerHTML;
}

/* 2. Transparent Header turns Solid Espresso on Scroll */
function initScrollNavbar() {
  const header = document.getElementById('site-header');
  if (!header) return;

  function onScroll() {
    if (window.scrollY > 50) {
      header.classList.add('is-scrolled');
    } else {
      header.classList.remove('is-scrolled');
    }
  }

  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();
}

/* 3. Mobile Menu Toggle */
function initMobileNavigation() {
  const toggleBtn = document.getElementById('mobile-menu-btn');
  const mobileMenu = document.getElementById('mobile-menu');
  const iconOpen = document.getElementById('menu-icon-open');
  const iconClose = document.getElementById('menu-icon-close');

  if (!toggleBtn || !mobileMenu) return;

  function setMenu(isOpen) {
    toggleBtn.setAttribute('aria-expanded', String(isOpen));
    mobileMenu.classList.toggle('hidden', !isOpen);
    if (iconOpen && iconClose) {
      iconOpen.classList.toggle('hidden', isOpen);
      iconClose.classList.toggle('hidden', !isOpen);
    }
  }

  toggleBtn.addEventListener('click', () => {
    const isOpen = toggleBtn.getAttribute('aria-expanded') === 'true';
    setMenu(!isOpen);
  });

  // Close mobile drawer when clicking an anchor link
  mobileMenu.querySelectorAll('a').forEach((link) => {
    link.addEventListener('click', () => setMenu(false));
  });

  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && toggleBtn.getAttribute('aria-expanded') === 'true') {
      setMenu(false);
      toggleBtn.focus();
    }
  });
}

/* 4. Hero Glass Form Demo Submission */
function initHeroTourForm() {
  const form = document.getElementById('hero-tour-form');
  const successPanel = document.getElementById('hero-tour-success');
  const nameField = document.getElementById('hero-visitor-name');
  const displaySpan = document.getElementById('hero-submitted-name');

  if (!form || !successPanel) return;

  form.addEventListener('submit', (e) => {
    e.preventDefault();
    if (!form.checkValidity()) {
      form.reportValidity();
      return;
    }

    if (displaySpan && nameField) {
      displaySpan.textContent = nameField.value.trim() || 'friend';
    }

    form.classList.add('hidden');
    successPanel.classList.remove('hidden');
  });
}

/* 5. FAQ Accordion */
function initFaqAccordion() {
  const faqButtons = document.querySelectorAll('[data-faq-btn]');
  if (!faqButtons.length) return;

  faqButtons.forEach((btn) => {
    btn.addEventListener('click', () => {
      const targetId = btn.getAttribute('aria-controls');
      const panel = document.getElementById(targetId);
      if (!panel) return;

      const isExpanded = btn.getAttribute('aria-expanded') === 'true';

      faqButtons.forEach((otherBtn) => {
        if (otherBtn !== btn) {
          otherBtn.setAttribute('aria-expanded', 'false');
          const otherPanel = document.getElementById(otherBtn.getAttribute('aria-controls'));
          if (otherPanel) otherPanel.classList.add('hidden');
        }
      });

      btn.setAttribute('aria-expanded', String(!isExpanded));
      panel.classList.toggle('hidden', isExpanded);
    });
  });
}
